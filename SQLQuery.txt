/*CREATE*/

CREATE TABLE author
(
	authorid     CHAR(7)         NOT NULL, 
	first        VARCHAR(25)     NOT NULL,
	last         VARCHAR(25)     NOT NULL, 
	biography    VARCHAR(150)    NULL,
	PRIMARY KEY(authorid)
);

CREATE TABLE publisher
(
	publisherid   CHAR(5)        NOT NULL,
	name          VARCHAR(25)    NOT NULL,
	street        VARCHAR(30)    NOT NULL,
	zip           CHAR(10)       NOT NULL,
	state         CHAR(5)        NOT NULL,
	country       VARCHAR(15)    NOT NULL,
	city          VARCHAR(15)    NOT NULL,
	email         VARCHAR(50)    NOT NULL,
	PRIMARY KEY (publisherid)
);

CREATE TABLE employee
(
	employeeid    CHAR(5)       NOT NULL,
	first         VARCHAR(15)   NOT NULL,  
	last          VARCHAR(25)   NOT NULL,
	position      VARCHAR(30)   NOT NULL,
	email         VARCHAR(35)   NOT NULL,
	PRIMARY KEY (employeeid)
);

CREATE TABLE customer
(
	customerid    CHAR(5)       NOT NULL,
	first         VARCHAR(15)   NOT NULL,  
	last          VARCHAR(25)   NOT NULL,
	street        VARCHAR(30)    NOT NULL,
	zip           CHAR(10)       NOT NULL,
	state         CHAR(5)        NOT NULL,
	country       VARCHAR(15)    NOT NULL,
	PRIMARY KEY (customerid)
);

CREATE TABLE order_
(
	orderid        CHAR(5)      NOT NULL,  
	total          FLOAT        NOT NULL,  
	odate          DATE         NOT NULL,  
	status         VARCHAR(10)  NOT NULL,
	shippingcost   FLOAT        NOT NULL,
	PRIMARY KEY (orderid)
);

CREATE TABLE promotion
(
	promotioncode                 CHAR(5)        NOT NULL,
	name                          VARCHAR(10)    NOT NULL,
	startperiod                   DATE           NOT NULL,
	endperiod                     DATE           NOT NULL,
	discountamount                INT            NOT NULL,
	PRIMARY KEY (promotioncode)  
);

CREATE TABLE book
(
	isbn            CHAR(5)        NOT NULL,
	title           VARCHAR(25)    NOT NULL,
	price           FLOAT          NOT NULL,
	stockquantity   INT            NOT NULL, 
	genre           VARCHAR(25)    NOT NULL,  
	publisherid     CHAR(5)        NOT NULL,
	PRIMARY KEY (isbn),
	FOREIGN KEY (publisherid) REFERENCES publisher(publisherid)
);

CREATE TABLE review
(
	reviewid     CHAR(5)        NOT NULL, 
	rating       INT            NOT NULL,  
	text         VARCHAR(150)   NULL,
	customerid    CHAR(5)       NOT NULL,
	PRIMARY KEY (reviewid),
	FOREIGN KEY (customerid) REFERENCES customer(customerid)
);

CREATE TABLE writes
(
   isbn            CHAR(5)        NOT NULL,
   authorid        CHAR(7)        NOT NULL,
   PRIMARY KEY (isbn, authorid),
   FOREIGN KEY (isbn) REFERENCES book(isbn),
   FOREIGN KEY (authorid) REFERENCES author(authorid)
);

CREATE TABLE has
(
   isbn            CHAR(5)        NOT NULL,
   reviewid        CHAR(5)        NOT NULL,
   PRIMARY KEY (isbn, reviewid),
   FOREIGN KEY (isbn) REFERENCES book(isbn),
   FOREIGN KEY (reviewid) REFERENCES review(reviewid)
);

CREATE TABLE includes
(
     orderid                CHAR(5)      NOT NULL,
     promotioncode          CHAR(5)      NOT NULL,
     PRIMARY KEY (orderid, promotioncode),
     FOREIGN KEY (orderid) REFERENCES order_(orderid),
     FOREIGN KEY (promotioncode) REFERENCES promotion(promotioncode)
);

CREATE TABLE order_item
(
     orderid         CHAR(5)      NOT NULL,
     isbn            CHAR(5)      NOT NULL,
     quantity        INT          NOT NULL,
     PRIMARY KEY (orderid, isbn),
     FOREIGN KEY (orderid) REFERENCES order_(orderid),
     FOREIGN KEY (isbn)    REFERENCES book(isbn) 
);

create table EmpManagesCustomers
(
    employeeid      CHAR(5)      NOT NULL,
    customerid      CHAR(5)      NOT NULL,
    PRIMARY KEY (employeeid, customerid),
    FOREIGN KEY (employeeid)    REFERENCES employee(employeeid),
    FOREIGN KEY (customerid)    REFERENCES customer(customerid)
);

create table EmpManagesOrders
(
    employeeid      CHAR(5)      NOT NULL,
    orderid         CHAR(5)      NOT NULL,
    PRIMARY KEY (employeeid, orderid),
    FOREIGN KEY (employeeid)    REFERENCES employee(employeeid),
    FOREIGN KEY (orderid) REFERENCES order_(orderid)
);

create table EmpManagesBooks
(
    employeeid      CHAR(5)      NOT NULL,
    isbn            CHAR(5)      NOT NULL,
    PRIMARY KEY (employeeid, isbn),
    FOREIGN KEY (employeeid)    REFERENCES employee(employeeid),
    FOREIGN KEY (isbn)    REFERENCES book(isbn)
);
